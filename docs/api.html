DummyPay API Reference

DummyPay is a simulated peer-to-peer payment platform. Registered users hold wallets of fictional money and send it to each other. This page documents every HTTP endpoint a signed-in user can call, with Python examples you can paste into a test script.

All money is fictional. No card network, bank or payment processor is ever contacted.

Overview

The API is served by Supabase. There are three kinds of endpoint, all under one base URL:

KindURL patternUsed for
Auth/auth/v1/…Sign up, sign in, tokens, passwords
Tables & views/rest/v1/<table>Reading your own data (and the few writes that are allowed)
Functions (RPC)/rest/v1/rpc/<function>Every operation that moves money or applies business rules
Edge Functions/functions/v1/<name>Adding payment methods, refreshing exchange rates
Base URL
https://vtimmqlnyoyxiwczfdwf.supabase.co
Why some calls use /rpc and some don't Table URLs read rows directly. Row Level Security (RLS) means you only ever get your own rows, and money tables are read-only. Anything that changes a balance goes through an /rpc/ function, which checks the caller, validates input and runs atomically on the server.

Authentication

Every request carries two headers:

HeaderValue
apikeyThe project's publishable (anon) key. It's public by design and identifies the project, not a user.
AuthorizationBearer <access_token>, the JWT returned when a user signs in. It identifies the user.

The server reads the caller's identity from the token. No endpoint accepts a user ID that decides who is acting, so a script can do exactly what that user could do in the app and nothing more.

  • Access tokens expire after 1 hour. Use the refresh token for longer test runs.
  • Only supported_currencies, sign-up and sign-in work without a user token.
Never use the service-role (secret) key for testing. It bypasses Row Level Security, so tests would exercise a path real users can't reach. Keep it out of scripts, repositories and CI logs.
Get the publishable key

Supabase dashboard → Project Settings → API Keys → publishable key. This is the same value as VITE_SUPABASE_ANON_KEY in the web app's .env.

Python client

The examples on this page use the small helper below. It needs only requests (pip install requests).

dummypay_client.py
import os
import uuid
import requests

BASE_URL = "https://vtimmqlnyoyxiwczfdwf.supabase.co"
ANON_KEY = os.environ["DUMMYPAY_ANON_KEY"]   # publishable key, never the secret key


class ApiError(Exception):
    """Raised for any non-2xx response. `code` is the DummyPay error code."""

    def __init__(self, status, body):
        self.status = status
        self.body = body
        msg = ""
        if isinstance(body, dict):
            # PostgREST errors use "message"; Edge Functions use "error";
            # Auth errors use "error_code" / "msg".
            msg = body.get("message") or body.get("error") or body.get("error_code") or body.get("msg") or ""
        self.code = msg.split(":", 1)[0].strip()
        super().__init__(f"{status} {self.code}: {body}")


class DummyPay:
    def __init__(self):
        self.http = requests.Session()
        self.http.headers["apikey"] = ANON_KEY
        self.refresh_token = None

    # ---- auth --------------------------------------------------------------
    def sign_in(self, email, password):
        r = self.http.post(f"{BASE_URL}/auth/v1/token", params={"grant_type": "password"},
                           json={"email": email, "password": password})
        return self._use_session(self._check(r))

    def refresh(self):
        r = self.http.post(f"{BASE_URL}/auth/v1/token", params={"grant_type": "refresh_token"},
                           json={"refresh_token": self.refresh_token})
        return self._use_session(self._check(r))

    def _use_session(self, session):
        self.http.headers["Authorization"] = f"Bearer {session['access_token']}"
        self.refresh_token = session["refresh_token"]
        self.user_id = session["user"]["id"]
        return session

    # ---- data --------------------------------------------------------------
    def rpc(self, function, **params):
        """POST /rest/v1/rpc/<function> with named parameters."""
        r = self.http.post(f"{BASE_URL}/rest/v1/rpc/{function}", json=params)
        return self._check(r)

    def select(self, table, **query):
        """GET /rest/v1/<table> with PostgREST query parameters."""
        r = self.http.get(f"{BASE_URL}/rest/v1/{table}", params=query)
        return self._check(r)

    def edge(self, name, body=None):
        """POST /functions/v1/<name>."""
        r = self.http.post(f"{BASE_URL}/functions/v1/{name}", json=body or {})
        return self._check(r)

    @staticmethod
    def _check(r):
        body = r.json() if r.content else None
        if not r.ok:
            raise ApiError(r.status_code, body)
        return body


if __name__ == "__main__":
    api = DummyPay()
    api.sign_in("alice@example.com", "alice-password")
    print(api.rpc("get_wallet_summary"))

Conventions

  • Money is in minor units (cents, agorot, pence) as integers. 1250 in USD means $12.50. Every supported currency has 2 decimal places.
  • Currencies: ILS, USD, EUR, GBP.
  • Amount side: send means the amount is what leaves your wallet. receive means it's what the recipient gets, and the server works out the sending amount.
  • Timestamps are ISO 8601 with a time zone, e.g. 2026-11-22T09:15:04.123+00:00.
  • RPC parameters are sent as a JSON object whose keys are the parameter names (p_currency, …). Optional parameters can be omitted.
  • RPC response shape depends on the function. Functions that return a table give a JSON array, even for one row. Functions that return a single record give a JSON object. Each endpoint below says which.
  • Table queries use PostgREST syntax: select=col1,col2, order=created_at.desc, filters like currency=eq.USD, paging with limit and offset.
  • Rounding: conversions round half-up to the nearest minor unit on the computed side.

Errors

Business-rule failures come back from /rest/v1/rpc/… as HTTP 400 with a PostgREST error body. The DummyPay error code is the text before the first colon in message:

{
  "code": "P0001",
  "details": null,
  "hint": null,
  "message": "INSUFFICIENT_FUNDS"
}
{
  "code": "P0001",
  "message": "QUOTE_EXPIRED: the quote has expired, please review the transfer again"
}

Edge Functions return {"error": "CODE"}, plus fields for validation errors.

HTTPMeaning
400Business rule rejected (P0001), or a malformed value such as an unknown currency (22P02: invalid input value for enum)
401 / 403Missing or expired token, or calling something your role isn't allowed to (42501)
404Unknown table or function, or wrong parameter names (PGRST202)
409Edge Function limit reached (LIMIT_REACHED)
Asserting on an error in a test
import pytest

def test_cannot_send_to_self(api):
    with pytest.raises(ApiError) as e:
        api.rpc("create_transfer_quote",
                p_recipient_id=api.user_id, p_sending_currency="USD",
                p_amount_minor=100, p_input_side="send")
    assert e.value.code == "SELF_TRANSFER"

All codes are listed in the error code index.

Rate limits

Per-user sliding-window limits protect recipient lookup and quoting. A rejected call fails with RATE_LIMITED and doesn't count towards the limit.

OperationLimit
lookup_recipient20 calls per rolling minute
create_transfer_quote30 successful quotes per rolling minute
add-payment-card10 cards stored per user
add-bank-account10 bank accounts stored per user
fx-refreshThe provider is contacted at most once per 30 minutes across all users

Supabase Auth also rate-limits sign-ins and sign-ups per IP address. Reuse one session per test user rather than signing in for every test.

Auth endpoints

These only need the apikey header (and the Bearer token where noted).

POST/auth/v1/signup

Registers a user. In the same database transaction, the server creates the profile and a wallet in the chosen currency, and credits the one-time welcome balance: ₪13,000.00 for ILS, or 5,000.00 for USD, EUR and GBP.

FieldTypeNotes
emailstringMust be unique
passwordstringAt least 8 characters
data.display_namestring2–50 characters, no leading or trailing spaces
data.preferred_currencystringILS | USD | EUR | GBP
r = requests.post(f"{BASE_URL}/auth/v1/signup", headers={"apikey": ANON_KEY}, json={
    "email": "carol@example.com",
    "password": "a-long-password",
    "data": {"display_name": "Carol", "preferred_currency": "EUR"},
})
print(r.status_code, r.json())   # includes access_token when email confirmation is off

An invalid display name or currency aborts the whole sign-up, and no user is created.

Bearer token

Returns the Auth user behind the token. It's useful to check that a token is still valid.

/recover emails a password-reset link. It responds the same way whether or not the address is registered, so it can't be used to discover accounts. PUT /user with a Bearer token sets a new password.

              json={"email": "alice@example.com"})

api.http.put(f"{BASE_URL}/auth/v1/user", json={"password": "new-long-password"})

POST/auth/v1/logout

Bearer token

Revokes the session's refresh token. Returns 204.

api.http.post(f"{BASE_URL}/auth/v1/logout")

Bearer tokenNo body

Transfers

Sending money takes three calls. The money only moves in the last one.

  1. lookup_recipient: find the registered user by email and get their user_id.
  2. create_transfer_quote: the server prices the transfer and locks in the rate for 60 seconds.
  3. execute_transfer: debits and credits both wallets in one atomic database transaction, records it in the ledger and notifies the recipient.
Money always arrives in the recipient's preferred currency. You choose which of your wallets to send from. The receiving currency is set by the recipient's profile, and the recipient gets a wallet in that currency automatically if they don't have one.

Payment methods

These are simulated. They aren't connected to any processor or bank, and they don't affect wallet balances. Full card numbers and CVVs are validated in memory and never stored. Account numbers and IBANs are stored masked only. Use test numbers, never real cards.

End-to-end test

A complete pytest example. It uses two pre-registered test accounts and checks that a transfer is atomic, idempotent and visible to both sides. Put the DummyPay client from above in dummypay_client.py next to it.

import os
import time
import uuid

import pytest
from dummypay_client import BASE_URL, ApiError, DummyPay


@pytest.fixture(scope="session")
def alice():
    api = DummyPay()
    api.sign_in(os.environ["ALICE_EMAIL"], os.environ["ALICE_PASSWORD"])
    return api


@pytest.fixture(scope="session")
def bob():
    api = DummyPay()
    api.sign_in(os.environ["BOB_EMAIL"], os.environ["BOB_PASSWORD"])
    return api


def balance(api, currency):
    rows = api.select("wallets", currency=f"eq.{currency}", select="balance_minor")
    return rows[0]["balance_minor"] if rows else 0


def test_transfer_moves_money_exactly_once(alice, bob):
    bob_profile = bob.select("profiles")[0]
    send_ccy = alice.select("profiles")[0]["preferred_currency"]
    recv_ccy = bob_profile["preferred_currency"]

    alice_before = balance(alice, send_ccy)
    bob_before = balance(bob, recv_ccy)

    found = alice.rpc("lookup_recipient", p_email=bob_profile["email"])[0]
    assert found["status"] == "found"

    quote = alice.rpc("create_transfer_quote",
                      p_recipient_id=found["user_id"], p_sending_currency=send_ccy,
                      p_amount_minor=100, p_input_side="send")

    key = str(uuid.uuid4())
    txn = alice.rpc("execute_transfer", p_quote_id=quote["quote_id"], p_idempotency_key=key)
    replay = alice.rpc("execute_transfer", p_quote_id=quote["quote_id"], p_idempotency_key=key)
    assert replay["id"] == txn["id"]

    assert balance(alice, send_ccy) == alice_before - quote["sending_amount_minor"]
    assert balance(bob, recv_ccy) == bob_before + quote["receiving_amount_minor"]

    notes = bob.select("notifications", transaction_id=f"eq.{txn['id']}")
    assert len(notes) == 1 and notes[0]["amount_minor"] == quote["receiving_amount_minor"]


def test_quote_cannot_be_reused_with_new_key(alice, bob):
    bob_email = bob.select("profiles")[0]["email"]
    rid = alice.rpc("lookup_recipient", p_email=bob_email)[0]["user_id"]
    ccy = alice.select("profiles")[0]["preferred_currency"]
    quote = alice.rpc("create_transfer_quote", p_recipient_id=rid, p_sending_currency=ccy,
                      p_amount_minor=100, p_input_side="send")
    alice.rpc("execute_transfer", p_quote_id=quote["quote_id"], p_idempotency_key=str(uuid.uuid4()))

    with pytest.raises(ApiError) as e:
        alice.rpc("execute_transfer", p_quote_id=quote["quote_id"], p_idempotency_key=str(uuid.uuid4()))
    assert e.value.code == "QUOTE_USED"


def test_balances_are_read_only(alice):
    wallet = alice.select("wallets")[0]
    r = alice.http.patch(f"{BASE_URL}/rest/v1/wallets",
                         params={"id": f"eq.{wallet['id']}"}, json={"balance_minor": 10**9})
    assert r.status_code in (401, 403)


def test_cannot_read_other_users_wallets(alice, bob):
    bob_wallet_ids = {w["id"] for w in bob.select("wallets")}
    alice_view = {w["id"] for w in alice.select("wallets")}
    assert not (bob_wallet_ids & alice_view)
Test data is permanent. Ledger rows can't be changed or deleted, so every test transfer stays in both users' history. Use dedicated test accounts and small amounts.
